cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(NeverDWebMCP LANGUAGES CXX)
  include(CTest)
endif()

if(NOT TARGET neverd_tool_web)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../common" common)
endif()
if(NOT TARGET neverd_shared)
  set(NEVERD_ENGINE_LIBRARY "" CACHE FILEPATH "Matching NeverD shared C ABI library")
  if(NOT EXISTS "${NEVERD_ENGINE_LIBRARY}")
    message(FATAL_ERROR "Set NEVERD_ENGINE_LIBRARY to a matching libneverd shared library")
  endif()
  add_library(neverd_shared SHARED IMPORTED GLOBAL)
  set_target_properties(neverd_shared PROPERTIES IMPORTED_LOCATION "${NEVERD_ENGINE_LIBRARY}")
  if(WIN32)
    set(NEVERD_ENGINE_IMPLIB "" CACHE FILEPATH "NeverD import library")
    if(NOT EXISTS "${NEVERD_ENGINE_IMPLIB}")
      message(FATAL_ERROR "Set NEVERD_ENGINE_IMPLIB to the matching NeverD import library")
    endif()
    set_target_properties(neverd_shared PROPERTIES IMPORTED_IMPLIB "${NEVERD_ENGINE_IMPLIB}")
  endif()
endif()

add_library(neverd_web_mcp_protocol STATIC Server.cpp Stdio.cpp Catalog.cpp)
target_link_libraries(neverd_web_mcp_protocol PUBLIC neverd_tool_json)
add_library(neverd_web_mcp_tools STATIC WebTools.cpp)
target_link_libraries(neverd_web_mcp_tools PUBLIC neverd_web_mcp_protocol neverd_tool_web)
add_executable(neverd-web-mcp Main.cpp)
target_link_libraries(neverd-web-mcp PRIVATE neverd_web_mcp_tools neverd_shared)
if(MINGW)
  target_link_options(neverd-web-mcp PRIVATE -municode)
endif()
set_target_properties(neverd-web-mcp PROPERTIES
  INSTALL_RPATH "$ORIGIN/../lib;@loader_path/../lib")
neverd_stage_json_notice(neverd-web-mcp)
install(TARGETS neverd-web-mcp RUNTIME DESTINATION bin)

if(BUILD_TESTING OR NEVERD_BUILD_TESTS)
  add_executable(neverd-web-mcp-protocol-tests tests/ProtocolTests.cpp)
  target_link_libraries(neverd-web-mcp-protocol-tests PRIVATE neverd_web_mcp_protocol)
  add_test(NAME NeverDWebMCPProtocol COMMAND neverd-web-mcp-protocol-tests)
  set_tests_properties(NeverDWebMCPProtocol PROPERTIES TIMEOUT 30 LABELS "web;mcp")
  add_executable(neverd-web-mcp-catalog-tests tests/CatalogTests.cpp)
  target_link_libraries(neverd-web-mcp-catalog-tests PRIVATE neverd_web_mcp_protocol)
  add_test(NAME NeverDWebMCPCatalog COMMAND neverd-web-mcp-catalog-tests)
  set_tests_properties(NeverDWebMCPCatalog PROPERTIES TIMEOUT 30 LABELS "web;mcp")
  add_executable(neverd-web-mcp-web-tests tests/WebTests.cpp)
  target_link_libraries(neverd-web-mcp-web-tests PRIVATE neverd_web_mcp_tools neverd_shared)
  add_test(NAME NeverDWebMCPWeb COMMAND neverd-web-mcp-web-tests)
  set_tests_properties(NeverDWebMCPWeb PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 LABELS "web;mcp")
  add_executable(neverd-web-mcp-process-tests tests/ProcessTests.cpp)
  target_link_libraries(neverd-web-mcp-process-tests PRIVATE neverd_web_mcp_protocol)
  add_dependencies(neverd-web-mcp-process-tests neverd-web-mcp)
  add_test(NAME NeverDWebMCPProcess COMMAND neverd-web-mcp-process-tests "$<TARGET_FILE:neverd-web-mcp>")
  set_tests_properties(NeverDWebMCPProcess PROPERTIES TIMEOUT 90 SKIP_RETURN_CODE 77 LABELS "web;mcp")
  add_test(NAME NeverDWebMCPClaude COMMAND neverd-web-mcp-process-tests "$<TARGET_FILE:neverd-web-mcp>" --claude)
  set_tests_properties(NeverDWebMCPClaude PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77 LABELS "web;mcp")
endif()

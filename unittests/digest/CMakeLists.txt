cmake_minimum_required(VERSION 3.20)
set(_neverd_root "${CMAKE_CURRENT_LIST_DIR}/../..")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(NeverDDigestTests LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  include(CTest)
  include("${_neverd_root}/cmake/NeverDMimalloc.cmake")
  if(LLVM_DIR)
    find_package(LLVM REQUIRED CONFIG)
  else()
    # Configure the recorded fork for this host. Only Support/TargetParser and
    # their dependencies are compiled, without the decompiler or code generators.
    set(LLVM_TARGETS_TO_BUILD "" CACHE STRING "" FORCE)
    set(LLVM_ENABLE_PROJECTS "" CACHE STRING "" FORCE)
    set(LLVM_INCLUDE_TESTS OFF CACHE BOOL "" FORCE)
    set(LLVM_INCLUDE_BENCHMARKS OFF CACHE BOOL "" FORCE)
    set(LLVM_INCLUDE_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LLVM_INCLUDE_TOOLS OFF CACHE BOOL "" FORCE)
    set(LLVM_INCLUDE_UTILS OFF CACHE BOOL "" FORCE)
    set(LLVM_ENABLE_RTTI ON CACHE BOOL "" FORCE)
    set(LLVM_ENABLE_ZLIB OFF CACHE BOOL "" FORCE)
    set(LLVM_ENABLE_ZSTD OFF CACHE BOOL "" FORCE)
    set(LLVM_ENABLE_LIBXML2 OFF CACHE BOOL "" FORCE)
    add_subdirectory("${_neverd_root}/third_party/llvm-project/llvm" llvm EXCLUDE_FROM_ALL)
    set(LLVM_INCLUDE_DIRS
      "${_neverd_root}/third_party/llvm-project/llvm/include"
      "${CMAKE_BINARY_DIR}/llvm/include")
  endif()
  separate_arguments(_llvm_definitions NATIVE_COMMAND "${LLVM_DEFINITIONS}")
  add_compile_options(${_llvm_definitions})
  include(FetchContent)
  FetchContent_Declare(googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.17.0.tar.gz
    URL_HASH SHA256=65fab701d9829d38cb77c14acdc431d2108bfdbf8979e40eb8ae567edf10b27c)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  set(INSTALL_GTEST OFF)
  FetchContent_MakeAvailable(googletest)
  include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/NeverDGoogleTest.cmake")
  neverd_configure_googletest()
  add_library(NeverDDigest STATIC "${_neverd_root}/lib/support/SHA256.cpp")
  target_include_directories(NeverDDigest PUBLIC "${_neverd_root}/include" ${LLVM_INCLUDE_DIRS})
  target_link_libraries(NeverDDigest PUBLIC LLVMSupport LLVMTargetParser)
endif()

include(GoogleTest)
if(NEVERD_DIGEST_REQUIRE_ARM64 AND NEVERD_DIGEST_REQUIRE_X86_64)
  message(FATAL_ERROR "Digest qualification requires exactly one native architecture")
endif()
set(NEVERD_DIGEST_BASELINE_ROOT "" CACHE PATH
  "Optional previous NeverD source tree for same-build native digest comparisons")
set(_modes Native Portable)
if(NEVERD_DIGEST_BASELINE_ROOT)
  if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    message(FATAL_ERROR "Digest baseline comparisons require the standalone digest build")
  endif()
  list(APPEND _modes Baseline)
endif()
foreach(_mode IN LISTS _modes)
  set(_owner NeverDDigest)
  set(_target NeverDDigestTests)
  set(_owner_root "${_neverd_root}")
  if(_mode STREQUAL "Portable")
    set(_owner NeverDPortableDigestTestOwner)
    set(_target NeverDPortableDigestTests)
  elseif(_mode STREQUAL "Baseline")
    set(_owner NeverDBaselineDigestTestOwner)
    set(_target NeverDBaselineDigestTests)
    set(_owner_root "${NEVERD_DIGEST_BASELINE_ROOT}")
  endif()
  if(NOT _mode STREQUAL "Native")
    add_library(${_owner} STATIC "${_owner_root}/lib/support/SHA256.cpp")
    target_include_directories(${_owner} PUBLIC "${_owner_root}/include" ${LLVM_INCLUDE_DIRS})
    target_link_libraries(${_owner} PUBLIC LLVMSupport LLVMTargetParser)
    target_compile_features(${_owner} PUBLIC cxx_std_20)
  endif()
  if(_mode STREQUAL "Portable")
    target_compile_definitions(${_owner} PRIVATE NEVERD_SHA256_PORTABLE)
  endif()
  add_executable(${_target} DigestTests.cpp ../lift/format/InputDigestTests.cpp)
  if(_mode STREQUAL "Native")
    target_sources(${_target} PRIVATE FeatureTests.cpp)
  endif()
  target_compile_features(${_target} PRIVATE cxx_std_20)
  foreach(_architecture ARM64 X86_64)
    if(NEVERD_DIGEST_REQUIRE_${_architecture})
      target_compile_definitions(${_target} PRIVATE NEVERD_DIGEST_REQUIRE_${_architecture})
    endif()
  endforeach()
  target_link_libraries(${_target} PRIVATE ${_owner} GTest::gtest_main)
  target_include_directories(${_target} PRIVATE "${_owner_root}/lib/support")
  neverd_link_mimalloc(${_target})
  if(_mode STREQUAL "Portable")
    target_compile_definitions(${_target} PRIVATE NEVERD_SHA256_PORTABLE)
  endif()
  if(TARGET NeverDUnitTests)
    add_dependencies(NeverDUnitTests ${_target})
  endif()
  gtest_discover_tests(${_target} DISCOVERY_MODE PRE_TEST
    TEST_PREFIX "${_target}." PROPERTIES TIMEOUT 120 LABELS ${_target})
  add_executable(${_target}Bench EXCLUDE_FROM_ALL Benchmark.cpp)
  target_compile_features(${_target}Bench PRIVATE cxx_std_20)
  target_include_directories(${_target}Bench PRIVATE "${_owner_root}/lib/support")
  target_link_libraries(${_target}Bench PRIVATE ${_owner})
  neverd_link_mimalloc(${_target}Bench)
endforeach()

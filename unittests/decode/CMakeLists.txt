cmake_minimum_required(VERSION 3.20)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  add_neverd_unittest(NeverDDecoderOptionsTests
    DecoderOptionsTests.cpp
    LINK_COMPONENTS Decode)
  if(NOT NEVERD_ENABLE_ICED)
    return()
  endif()
else()
  project(NeverDIcedAdapterTests LANGUAGES C CXX)
  get_filename_component(PROJECT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
  set(CMAKE_CXX_STANDARD 20)
  enable_testing()
  set(CAPSTONE_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(CAPSTONE_BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
  set(CAPSTONE_BUILD_CSTOOL OFF CACHE BOOL "" FORCE)
  set(CAPSTONE_BUILD_CSTEST OFF CACHE BOOL "" FORCE)
  set(CAPSTONE_BUILD_LEGACY_TESTS OFF CACHE BOOL "" FORCE)
  set(CAPSTONE_BUILD_MACOS_THIN ON CACHE BOOL "" FORCE)
  set(CAPSTONE_ARCHITECTURE_DEFAULT OFF CACHE BOOL "" FORCE)
  set(CAPSTONE_X86_SUPPORT ON CACHE BOOL "" FORCE)
  set(CAPSTONE_X86_SCAN_ADAPTER ON CACHE BOOL "" FORCE)
  add_subdirectory("${PROJECT_SOURCE_DIR}/third_party/capstone" capstone)
  include("${PROJECT_SOURCE_DIR}/cmake/NeverDIced.cmake")
endif()
add_executable(neverd-iced-parity X86IcedScanTests.cpp)
target_include_directories(neverd-iced-parity PRIVATE "${PROJECT_SOURCE_DIR}/lib/decode")
target_compile_definitions(neverd-iced-parity PRIVATE
  NEVERD_ICED_TEST_DATA="${PROJECT_SOURCE_DIR}/third_party/iced/src/UnitTests/Intel/Decoder")
target_link_libraries(neverd-iced-parity PRIVATE NeverDX86IcedScan capstone_static)
add_test(NAME NeverDIcedScanParity COMMAND neverd-iced-parity)
